cmake_minimum_required(VERSION 3.22)

# Packaging presets use the same configure/build commands as desktop presets.
# Gradle's native CMake invocation does not enable these packaging switches.
option(CRYSTAL_BUILD_APK "Build an Android APK through Gradle" OFF)
option(CRYSTAL_BUILD_PORTABLE "Build portable Linux through Docker" OFF)
if(CRYSTAL_BUILD_APK)
  project(CrystalAndroidPackaging LANGUAGES NONE)
  include("${CMAKE_CURRENT_SOURCE_DIR}/AndroidBuild.cmake")
  return()
elseif(CRYSTAL_BUILD_PORTABLE)
  project(CrystalPortablePackaging LANGUAGES NONE)
  include("${CMAKE_CURRENT_SOURCE_DIR}/PortableBuild.cmake")
  return()
endif()

# Discover the host's vcpkg before project() loads the compiler/toolchain.
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  if("$ENV{VCPKG_ROOT}" STREQUAL "")
    # Machine-independent locations first; retain the old D: location as fallback.
    foreach(candidate IN ITEMS
        "${CMAKE_CURRENT_SOURCE_DIR}/.tools/vcpkg"
        "${CMAKE_CURRENT_SOURCE_DIR}/../vcpkg"
        "$ENV{USERPROFILE}/vcpkg"
        "$ENV{HOME}/vcpkg"
        "D:/vcpkg")
      if(EXISTS "${candidate}/scripts/buildsystems/vcpkg.cmake")
        set(ENV{VCPKG_ROOT} "${candidate}")
        break()
      endif()
    endforeach()
  endif()
  if(NOT "$ENV{VCPKG_ROOT}" STREQUAL "")
    set(CMAKE_TOOLCHAIN_FILE "$ENV{VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake" CACHE FILEPATH "vcpkg toolchain")
  endif()
endif()
if(CMAKE_HOST_SYSTEM_NAME STREQUAL "Windows" AND
   (NOT DEFINED CMAKE_TOOLCHAIN_FILE OR NOT EXISTS "${CMAKE_TOOLCHAIN_FILE}"))
  message(FATAL_ERROR "Install vcpkg in <project>/.tools/vcpkg, next to the project, or in %USERPROFILE%/vcpkg; otherwise set VCPKG_ROOT to its installation directory. No preset edits are needed.")
endif()
if(DEFINED ENV{VCPKG_DEFAULT_TRIPLET} AND NOT DEFINED VCPKG_TARGET_TRIPLET)
  set(VCPKG_TARGET_TRIPLET "$ENV{VCPKG_DEFAULT_TRIPLET}" CACHE STRING "vcpkg triplet")
endif()
set(VCPKG_FEATURE_FLAGS "versions")
if(NOT DEFINED VCPKG_BUILD_TYPE)
  set(VCPKG_BUILD_TYPE "release")
endif()
set(VCPKG_BUILD_TYPE "${VCPKG_BUILD_TYPE}" CACHE STRING "vcpkg build type")
option(OTCLIENT_BUILD_CLIENT "Build client executable" ON)
option(OTCLIENT_BUILD_TESTS "Build unit tests" OFF)
set(VCPKG_MANIFEST_FEATURES "${VCPKG_MANIFEST_FEATURES}" CACHE STRING "vcpkg manifest features")

if(VCPKG_TARGET_ANDROID)
  include("cmake/vcpkg_android.cmake")
  set(Android_INCLUDES ${Android_INCLUDES} ${CMAKE_SOURCE_DIR}/android/app/libs/include)
  set(Android_LIBRARIES ${Android_LIBRARIES} ${CMAKE_SOURCE_DIR}/android/app/libs/lib/${CMAKE_ANDROID_ARCH_ABI})
  include_directories(${Android_INCLUDES})
endif()
if(CMAKE_VERSION VERSION_GREATER_EQUAL 3.19)
  include("cmake/SharedBuildCache.cmake")
endif()

project(otclient)
if(WIN32)
  include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/WindowsArchitecture.cmake")
endif()
list(APPEND CMAKE_MODULE_PATH ${CMAKE_CURRENT_SOURCE_DIR}/cmake)
include(MessageColors)
include(LoggingHelper)
include(GNUInstallDirs)

if(NOT DEFINED WASM)
  if(CMAKE_SYSTEM_NAME STREQUAL "Emscripten" OR CMAKE_CXX_COMPILER_ID MATCHES "Emscripten")
    set(WASM ON CACHE BOOL "Build WebAssembly" FORCE)
  else()
    set(WASM OFF CACHE BOOL "Build WebAssembly" FORCE)
  endif()
endif()
if(WASM)
  set(THREADS_PREFER_PTHREAD_FLAG ON)
  add_compile_options(-pthread -matomics -mbulk-memory)
  add_link_options(-pthread -matomics -mbulk-memory)
endif()
option(OPTIONS_ENABLE_CCACHE "Enable ccache" OFF)
option(OPTIONS_ENABLE_SCCACHE "Use sccache" OFF)
option(OPTIONS_ENABLE_IPO "Enable interprocedural optimization" ON)
if(OPTIONS_ENABLE_CCACHE)
  find_program(CCACHE ccache)
  if(CCACHE)
    set(CMAKE_CXX_COMPILER_LAUNCHER ${CCACHE})
  endif()
endif()
if(OPTIONS_ENABLE_SCCACHE)
  find_program(SCCACHE_PATH sccache)
  if(SCCACHE_PATH)
    set(CMAKE_C_COMPILER_LAUNCHER ${SCCACHE_PATH})
    set(CMAKE_CXX_COMPILER_LAUNCHER ${SCCACHE_PATH})
  endif()
endif()
if(OPTIONS_ENABLE_IPO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT ipo_supported OUTPUT ipo_error)
  if(ipo_supported)
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION TRUE)
  endif()
endif()
add_subdirectory(src)
if(OTCLIENT_BUILD_TESTS AND NOT ANDROID)
  if(NOT "tests" IN_LIST VCPKG_MANIFEST_FEATURES)
    message(FATAL_ERROR "Unit tests require VCPKG_MANIFEST_FEATURES=tests.")
  endif()
  enable_testing()
  add_subdirectory(tests)
endif()
